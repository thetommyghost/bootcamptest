> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. Phase 1 already shipped at deploy `1c23251c`.

**Goal:** Make team notes first-class citizens in the Bootcamp Scout simulation. Notes must be visible in UI, syncable across browsers via Cloudflare KV, threadable as per-zone discussions, browsable in a side panel, and exportable as Markdown — without breaking the polished Round 2 game.

**Architecture:**
1. **Phase 2 (UI-only, no backend):** render the localStorage notes that already exist in 4 UI surfaces (canvas pins, mini-map dots, briefing thread, HUD count). Shippable standalone.
2. **Phase 3 (Cloudflare KV backend):** wrap the localStorage layer with a fetch-based sync client. KV namespace `kawasist_bootcamp_notes`. Pages Function at `docs/functions/api/notes/index.js`. Polling every 30s while page visible. LocalStorage stays as offline-cache.
3. **Phase 4 (discussion threads):** add `reply_to` to schema, tabbed briefing (Overview · Discussion), inline reply compose, author chips with deterministic color hash.
4. **Phase 5 (browser + export):** click HUD pill → side panel with filter/search/export. Markdown round-trip.
5. **Phase 6 (close-out):** Lighthouse gate, smoke regression, deploy, STATUS.md + daily note + state log.

**Tech Stack:** Vanilla DOM/CSS/inline-SVG (no frameworks). Cloudflare Pages Functions (serverless JS). Cloudflare KV (key-value store). Playwright for smoke tests. `wrangler` CLI for KV + deploys.

```js
  const layer = document.querySelector('.note-pins');
  if (!layer) return;

  for (const n of notesState.notes){

    pin.setAttribute('role', 'button');
    pin.setAttribute('tabindex', '0');

    pin.addEventListener('mouseenter', (e) => showNotePinTip(pin, n));
    pin.addEventListener('mouseleave', hideNotePinTip);

let _tipEl = null;
function showNotePinTip(anchor, note){
  hideNotePinTip();

function hideNotePinTip(){
  if (_tipEl && _tipEl.parentNode) _tipEl.parentNode.removeChild(_tipEl);
  _tipEl = null;
}
function escapeHtml(s){
  return String(s).replace(/[&<>"']/g, c => ({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
  }[c]));
}
```

```js
function renderBriefingNotes(stationId){
  const overlay = $('roomOverlay');
  if (!overlay) return;
  const body = overlay.querySelector('.room-body, .room-body-inner, .room-content');
  if (!body) return;
  let host = body.querySelector('.notes-thread');
  if (!host){
    host = document.createElement('section');
    host.className = 'notes-thread';
    host.innerHTML = '<h4>NOTES</h4><div class="notes-thread-list"></div>';
    body.appendChild(host);
  }
  const list = host.querySelector('.notes-thread-list');
  list.innerHTML = '';
  const notes = notesState.notes
    .filter(n => !n.deleted_at && n.zone === stationId)
    .sort((a, b) => (a.created_at || 0) - (b.created_at || 0));
  if (notes.length === 0){
    list.innerHTML = '<div class="empty">No notes yet — press N to drop one.</div>';
    return;
  }
  for (const n of notes){
    const row = document.createElement('div');
    row.className = 'note-row';
    row.dataset.noteId = n.id;
    const when = new Date(n.created_at || Date.now());
    const whenStr = when.toLocaleString('en-GB', { day:'2-digit', month:'short', hour:'2-digit', minute:'2-digit' });
    row.innerHTML = `
      <div>
        <time datetime="${when.toISOString()}">${escapeHtml(whenStr)}</time>
        <span class="note-title">${escapeHtml(n.title || 'Untitled')}</span>
      </div>
      <p class="note-body">${escapeHtml(n.body || '')}</p>
      <div class="note-row-actions">
        <button data-act="edit">Edit</button>
        <button data-act="delete">Delete</button>
      </div>
    `;
    row.querySelector('[data-act="edit"]').addEventListener('click', () => openNoteEditor(n));
    row.querySelector('[data-act="delete"]').addEventListener('click', () => {
      if (!confirm('Delete this note?')) return;
      n.deleted_at = Date.now();
      saveNotesLocal();
      renderAllNotesUI();
    });
    list.appendChild(row);
  }
}
```

```js
    .replace(/[^a-z0-9_-]/g, '').slice(0, 64) || 'edward-said';
```

```js
function authorColorHash(name){
  let h = 0;
  for (let i = 0; i < (name||'').length; i++) h = ((h << 5) - h + name.charCodeAt(i)) | 0;
  const palette = ['#c08bd8','#7eb4ec','#7ed3c4','#ecc97e','#ec8e8e','#9ce07e','#e69ec8','#a896d6'];
  return palette[Math.abs(h) % palette.length];
}

    .replace(/\[([^\]]+)\]\((https?:\/\/[^)]+)\)/g, '<a href="$2" target="_blank" rel="noopener">$1</a>')
```

```js
function renderSidePanel(){
  const panel = document.getElementById('notesSidePanel');
  if (!panel) return;
  const search = (panel.querySelector('.nsp-search').value || '').toLowerCase();
  const zoneFilter = panel.querySelector('.nsp-zone-filter').value;
  const authorFilter = panel.querySelector('.nsp-author-filter').value;
  // Build filter option lists
  const allNotes = notesState.notes.filter(n => !n.deleted_at);
  const zones = [...new Set(allNotes.map(n => n.zone).filter(Boolean))].sort();
  const authors = [...new Set(allNotes.map(n => n.author || 'Anonymous'))].sort();
  const zoneSel = panel.querySelector('.nsp-zone-filter');
  const authorSel = panel.querySelector('.nsp-author-filter');
  if (zoneSel.dataset.populated !== '1' || zoneSel.options.length - 1 !== zones.length){
    zoneSel.innerHTML = '<option value="">All zones</option>' + zones.map(z => `<option value="${escapeHtml(z)}">${escapeHtml(z)}</option>`).join('');
    zoneSel.value = zoneFilter;
    zoneSel.dataset.populated = '1';
  }
  if (authorSel.dataset.populated !== '1' || authorSel.options.length - 1 !== authors.length){
    authorSel.innerHTML = '<option value="">All authors</option>' + authors.map(a => `<option value="${escapeHtml(a)}">${escapeHtml(a)}</option>`).join('');
    authorSel.value = authorFilter;
    authorSel.dataset.populated = '1';
  }
  // Apply filters
  let filtered = allNotes;
  if (search) filtered = filtered.filter(n =>
    (n.title || '').toLowerCase().includes(search) ||
    (n.body  || '').toLowerCase().includes(search)
  );
  if (zoneFilter) filtered = filtered.filter(n => n.zone === zoneFilter);
  if (authorFilter) filtered = filtered.filter(n => (n.author || 'Anonymous') === authorFilter);
  // Render
  const list = panel.querySelector('.nsp-list');
  panel.querySelector('.nsp-count').textContent = String(filtered.length);
  if (filtered.length === 0){
    list.innerHTML = '<div style="color:rgba(255,255,255,.4); padding:20px; text-align:center;">No notes match.</div>';
    return;
  }
  const byZone = {};
  for (const n of filtered){
    const z = n.zone || '_free';
    (byZone[z] = byZone[z] || []).push(n);
  }
  list.innerHTML = '';
  for (const z of Object.keys(byZone).sort()){
    const group = document.createElement('div');
    group.className = 'nsp-zone-group';
    group.innerHTML = `<h3>${escapeHtml(z)}</h3>`;
    for (const n of byZone[z].sort((a, b) => (b.created_at || 0) - (a.created_at || 0))){
      const author = n.author || 'Anonymous';
      const color = authorColorHash(author);
      const item = document.createElement('div');
      item.className = 'nsp-note';
      item.style.setProperty('--author-color', color);

        .toLocaleString('en-GB', { day:'2-digit', month:'short', hour:'2-digit', minute:'2-digit' });
      item.innerHTML = `
        <div class="nsp-note-title">${escapeHtml(n.title || 'Untitled')}</div>
```

```js
function exportMarkdown(){

      });

      const replies = allNotes.filter(r => r.reply_to === n.id)
        .sort((a,b) => (a.created_at || 0) - (b.created_at || 0));
      for (const r of replies){
        const rwhen = new Date(r.created_at || Date.now()).toLocaleString('en-GB', {

        });
        md += `  - **${r.author || 'Anonymous'} · ${rwhen}** — ${r.body.replace(/\n/g, ' ')}\n`;
      }
    }
    md += `\n`;
  }
  const blob = new Blob([md], { type: 'text/markdown;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
```

```js
async function importMarkdownFile(file){
  const txt = await file.text();
  // Parse: ## <zone> followed by - **author · date** — title \n  body
  const zoneRe = /^##\s+(.+)$/gm;
  const noteRe = /^-\s+\*\*([^·]+?)\s+·\s+([^*]+?)\*\*(?:\s+★)?(?:\s+—\s+(.+?))?$/;
  const lines = txt.split('\n');
  let currentZone = null;
  let current = null;
  const imported = [];
  for (const line of lines){
    const zm = line.match(/^##\s+(.+)$/);
    if (zm){ currentZone = zm[1].trim(); continue; }
    const nm = line.match(noteRe);
    if (nm && currentZone){
      if (current) imported.push(current);
      current = {
        id: 'imp_' + Date.now() + '_' + Math.random().toString(36).slice(2, 8),
        zone: currentZone,
        pos: { x: 50, y: 50 },  // default — no pos info in MD
        author: nm[1].trim(),
        title: (nm[3] || '').trim(),
        body: '',
        created_at: Date.parse(nm[2]) || Date.now(),
        updated_at: Date.now(),
      };
      continue;
    }
    if (current && /^\s{2,}\S/.test(line)){
      current.body = (current.body + '\n' + line.replace(/^\s{2}/, '')).trim();
    }
  }
  if (current) imported.push(current);
  if (imported.length === 0){
    alert('No notes parsed from file.');
    return;
  }
  if (!confirm(`Import ${imported.length} notes? They will be merged with new IDs (no conflicts).`)){
    return;
  }
  for (const n of imported){
    await saveNote(n);
  }
  alert(`Imported ${imported.length} notes.`);
  renderSidePanel();
}
```
